```
#' Simulate Data and Plot Distributions
#'
#' This function simulates data based on specified parameters and plots the distributions.
#'
#' @param n_samples Number of samples to simulate.
#' @param mean Mean value for the distribution.
#' @param sd Standard deviation for the distribution.
#' @param dist_type Type of distribution ('normal', 'uniform').
#' @return A ggplot object showing the simulated data distribution.
#'
#' @examples
#' temp_simulation(n_samples = 1000, mean = 5, sd = 2, dist_type = "normal")
temp_simulation <- function(n_samples, mean, sd, dist_type) {
  if (dist_type == "normal") {
    data <- rnorm(n_samples, mean = mean, sd = sd)
  } else if (dist_type == "uniform") {
    data <- runif(n_samples, min = mean - sd, max = mean + sd)
  } else {
    stop("Unsupported distribution type")
  }

  ggplot(data.frame(x = data), aes(x = x)) +
    geom_histogram(bins = 30, fill = "lightblue", color = "black") +
    theme_minimal() +
    labs(title = paste("Simulated Data Distribution (n =", n_samples, ")"))
}
```
